[ X1 ] // LIVE SIMULATION

Simulate before you solder.

Every tutorial ships with a circuit you can run in the browser. Here’s the classic first sketch — press Run and watch GPIO 2 come alive.

blink.ino 
// Your first ESP32 program
const int LED = 2;   // onboard LED → GPIO 2

void setup() {
  pinMode(LED, OUTPUT);
  Serial.begin(115200);
}

void loop() {
  digitalWrite(LED, HIGH);
  Serial.println("LED ON");
  delay(600);
  digitalWrite(LED, LOW);
  Serial.println("LED OFF");
  delay(600);
}
